using System;
using System.Collections.Generic;
using System.Linq;

namespace Sample;

class Widget
{
    public int Value;
}

class Program
{
    static void Main()
    {
        int oldName = 42;
        Console.WriteLine(oldName * 2);
        PrintGreeting();

        var widgets = new List<Widget> { new Widget { Value = 1 }, new Widget { Value = 2 } };
        var renamedResultsList = new List<int>();
        foreach (var renamedWidgetVar in widgets)
        {
            if (renamedWidgetVar.Value > 0)
            {
                renamedResultsList.Add(renamedWidgetVar.Value * 2);
            }
        }
    }

    static void PrintGreeting()
    {
        Console.WriteLine("Hello");
    }
}
