using System;
using System.Collections.Generic;
using System.Linq;

namespace Sample;

class Widget
{
    public int Value;
}

class Program
{
    static void Main()
    {
        int oldName = 42;
        Console.WriteLine(oldName * 2);

        var widgets = new List<Widget> { new Widget { Value = 1 }, new Widget { Value = 2 } };
        var results = (from widget in widgets
                       where widget.Value > 0
                       select widget.Value * 2).ToList();
    }
}
